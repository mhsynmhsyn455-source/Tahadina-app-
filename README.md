function pad(n){return n<10?'0'+n:''+n;}
function keyOf(d){return d.getFullYear()+'-'+pad(d.getMonth()+1)+'-'+pad(d.getDate());}
function parseKey(k){var a=k.split('-').map(Number); return Date.UTC(a[0],a[1]-1,a[2]);}
function diffDays(a,b){return Math.round((parseKey(a)-parseKey(b))/86400000);}
function addDays(k,n){var t=new Date(parseKey(k)+n*86400000); return t.getUTCFullYear()+'-'+pad(t.getUTCMonth()+1)+'-'+pad(t.getUTCDate());}
function today(){return keyOf(new Date());}
var WD=['ح','ن','ث','ر','خ','ج','س'];
function wd(k){return WD[new Date(parseKey(k)).getUTCDay()];}
function dom(k){return Number(k.slice(8));}
function streakEnding(set,t){
  var c=0, k=set.has(t)?t:addDays(t,-1);
  while(set.has(k)){c++; k=addDays(k,-1);}
  return c;
}
function bestStreak(set){
  var ks=Array.from(set).sort(), best=0, cur=0, prev=null;
  ks.forEach(function(k){ cur=(prev&&diffDays(k,prev)===1)?cur+1:1; if(cur>best)best=cur; prev=k; });
  return best;
}

function embedOf(u){
  var m=u.match(/[?&]v=([\w-]+)/); if(m) return 'https://www.youtube.com/embed/'+m[1];

    }},last?'عرض النتيجة':'التالي'):null);
}

function viewBadges(){
  var st=stats(), got=BADGES.filter(function(b){return b.ok(st)}).length;
  var steps=[3,7,14,30], next=steps.filter(function(n){return st.best<n})[0];
  var nextCard=next
    ?h('div',{class:'card stack-s'},
        h('div',{class:'row'},h('div',{class:'tile t1'},icon('flame',24)),
          h('div',{class:'grow'},h('div',{class:'muted',text:'الشارة القادمة'}),h('div',{style:'font-size:18px;font-weight:700',text:ar(next)+' أيام متتالية'})),
          h('span',{style:'font-weight:700',text:ar(st.best)+' / '+ar(next)})),
        h('div',{class:'bar'},h('i',{style:'width:'+Math.round(st.best/next*100)+'%'})))
    :h('div',{class:'card center',text:'أكملت كل شارات السلاسل. ما شاء الله!'});
  var grid=h('div',{class:'badges'});
  BADGES.forEach(function(b){
    var on=b.ok(st);
    grid.append(h('div',{class:'badge'+(on?'':' off')},h('div',{class:'tile'},icon(on?b.ic:'lock',on?26:24)),
      h('span',{text:b.t}),on?null:h('span',{class:'muted',text:'مقفلة'})));
  });
  return h('div',{class:'stack'},
    h('div',null,h('h1',{class:'h1',text:'شاراتي'}),h('div',{class:'muted',text:ar(got)+' شارات من '+ar(BADGES.length)})),
    nextCard,grid);
}
